cmake_minimum_required(VERSION 3.20)
project(wctSurfaceCurve LANGUAGES CXX)
set(MAYA_LOCATION "C:/Program Files/Autodesk/Maya2024" CACHE PATH "Maya 2024 installation")
set(MAYA_DEVKIT "${MAYA_LOCATION}" CACHE PATH "Root containing include/maya and lib")
find_path(MAYA_INCLUDE maya/MPxNode.h PATHS "${MAYA_DEVKIT}/include" "${MAYA_LOCATION}/include" NO_DEFAULT_PATH REQUIRED)
find_library(MAYA_OPENMAYA OpenMaya PATHS "${MAYA_DEVKIT}/lib" "${MAYA_LOCATION}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(MAYA_FOUNDATION Foundation PATHS "${MAYA_DEVKIT}/lib" "${MAYA_LOCATION}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(wctSurfaceCurve SHARED src/WctSurfaceCurveNode.cpp src/CurveEdgeCutsCommand.cpp)
target_compile_features(wctSurfaceCurve PRIVATE cxx_std_17)
target_include_directories(wctSurfaceCurve PRIVATE "${MAYA_INCLUDE}")
target_link_libraries(wctSurfaceCurve PRIVATE "${MAYA_OPENMAYA}" "${MAYA_FOUNDATION}")
if(WIN32)
 target_compile_definitions(wctSurfaceCurve PRIVATE NT_PLUGIN REQUIRE_IOSTREAM NOMINMAX)
 target_compile_options(wctSurfaceCurve PRIVATE /O2 /EHsc /utf-8)
 target_link_options(wctSurfaceCurve PRIVATE /EXPORT:initializePlugin /EXPORT:uninitializePlugin)
 set_target_properties(wctSurfaceCurve PROPERTIES PREFIX "" SUFFIX ".mll" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
set_target_properties(wctSurfaceCurve PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/dist")
